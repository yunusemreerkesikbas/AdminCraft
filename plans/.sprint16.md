### 7. TenantContextService - Tenant Selection
**File:** `storefront/.../core/tenant/tenant-context.service.ts`

**Add:**
```typescript
private _selectedTenant$ = new BehaviorSubject<Tenant | null>(null);
public selectedTenant$ = this._selectedTenant$.asObservable();

private _tenantModules$ = new BehaviorSubject<string[]>([]);
public tenantModules$ = this._tenantModules$.asObservable();

selectTenant(tenant: Tenant): void {
  this.setCurrentTenant(tenant);
  this._selectedTenant$.next(tenant);

  // SessionStorage (cleared when tab closes)
  sessionStorage.setItem('selectedTenantId', String(tenant.id));

  this.loadTenantModules(tenant.id);
}

private loadTenantModules(tenantId: number): void {
  this._http.get<ApiResponse<TenantModule[]>>(`/api/tenants/${tenantId}/modules`)
    .subscribe(response => {
      const moduleCodes = response.data
        .filter(m => m.status === 'enabled')
        .map(m => m.moduleCode);
      this._tenantModules$.next(moduleCodes);
    });
}

clearTenantSelection(): void {
  this._selectedTenant$.next(null);
  this._tenantModules$.next([]);
  sessionStorage.removeItem('selectedTenantId');
  localStorage.removeItem('tenantId');
}
```

### 9. Layout - Tenant Selector UI
**File:** `storefront/.../layout/layouts/vertical/classy/classy.component.ts`

**Add:**
```typescript
export class ClassyComponent implements OnInit {
  user: User;
  isSuperAdmin = false;
  tenantOptions: SpaSelectOption<number>[] = [];
  selectedTenantId: number | null = null;
  private tenants: Tenant[] = [];

  ngOnInit(): void {
    this._userService.user$.subscribe(user => {
      this.user = user;
      this.isSuperAdmin = user.role === 'SUPER_ADMIN';

      if (this.isSuperAdmin) {
        this.loadTenants();
        this.restoreLastSelectedTenant();
      }
    });

    this._tenantContext.selectedTenant$.subscribe(tenant => {
      this.selectedTenantId = tenant?.id || null;
    });
  }

  private loadTenants(): void {
    this._tenantService.getAllTenants().subscribe(tenants => {
      this.tenants = tenants;
      this.tenantOptions = tenants.map(t => ({
        value: t.id,
        label: `${t.name} (${t.subdomain})`
      }));
    });
  }

  private restoreLastSelectedTenant(): void {
    const savedId = sessionStorage.getItem('selectedTenantId');
    if (savedId) {
      const tenant = this.tenants.find(t => t.id === +savedId);
      if (tenant) this._tenantContext.selectTenant(tenant);
    }
  }

  onTenantChange(tenantId: number | null): void {
    if (!tenantId) {
      this._tenantContext.clearTenantSelection();
      return;
    }
    const tenant = this.tenants.find(t => t.id === tenantId);
    if (tenant) this._tenantContext.selectTenant(tenant);
  }
}
```

### 11. Navigation Service - Dynamic Filtering
**File:** `storefront/.../mock-api/common/navigation/api.ts`

**Add Filtering Logic:**
```typescript
constructor(
  private _httpClient: HttpClient,
  private _tenantContext: TenantContextService
) {
  // Re-filter navigation when tenant modules change
  this._tenantContext.tenantModules$.subscribe(modules => {
    this.filterNavigationByModules(modules);
  });
}

private filterNavigationByModules(enabledModules: string[]): void {
  const filtered = {
    default: this.filterItems(defaultNavigation, enabledModules),
    compact: this.filterItems(compactNavigation, enabledModules),
    futuristic: this.filterItems(futuristicNavigation, enabledModules),
    horizontal: this.filterItems(horizontalNavigation, enabledModules)
  };
  this._navigation.next(filtered);
}

private filterItems(items: FuseNavigationItem[], modules: string[]): FuseNavigationItem[] {
  return items
    .map(item => {
      // Platform module (requiredModule === null) → always show
      if (!item.requiredModule) {
        return {
          ...item,
          children: item.children ? this.filterItems(item.children, modules) : undefined
        };
      }

      // Tenant module → show if enabled
      if (modules.includes(item.requiredModule)) {
        return {
          ...item,
          children: item.children ? this.filterItems(item.children, modules) : undefined
        };
      }

      return null; // Module not enabled → hide
    })
    .filter(item => item !== null);
}
```

### 13. Tenant Interceptor - Early Blocking
**File:** `storefront/.../core/tenant/tenant.interceptor.ts`

**Update:**
```typescript
export const tenantInterceptor: HttpInterceptorFn = (req, next) => {
  const tenantContext = inject(TenantContextService);
  const userService = inject(UserService);

  // Skip platform endpoints
  const platformEndpoints = ['/api/tenants', '/api/provisioning', '/actuator'];
  if (platformEndpoints.some(ep => req.url.includes(ep))) {
    return next(req);
  }

  // SUPER_ADMIN + no tenant selected → block
  const user = userService.user;
  if (user?.role === 'SUPER_ADMIN' && !tenantContext.getCurrentTenantId()) {
    console.warn('No tenant selected, request blocked:', req.url);
    return throwError(() => new Error('Please select a tenant'));
  }

  // Add X-Tenant-ID header
  const tenantId = tenantContext.getCurrentTenantId();
  if (tenantId) {
    const cloned = req.clone({
      setHeaders: { 'X-Tenant-ID': String(tenantId) }
    });
    return next(cloned);
  }

  return next(req);
};
```
