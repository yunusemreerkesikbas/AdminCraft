## =' Technical Patterns

### Signal Pattern

```typescript

readonly data: Signal<Type> = this.#dataSig.asReadonly();

```

### Guard Pattern

```typescript
export const roleGuard = (allowedRoles: string[]): CanActivateFn => {
    return (route) => {
        const user = inject(UserService).user();
        return allowedRoles.includes(user?.role);
    };
};
```

### Navigation Filter Pattern

```typescript
if (item.requiredRole && userRole !== item.requiredRole) return null;
if (item.excludedRoles?.includes(userRole)) return null;
```
