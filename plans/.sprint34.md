## Phase 1 Guide: Yeni i18n Modülü Ekleme

### 1. Migration Oluştur

```sql
-- db/tenant/{module}/V{N}__{entity}_i18n.sql
CREATE TABLE {entity}_i18n (
    id BIGINT AUTO_INCREMENT PRIMARY KEY,
    uid VARCHAR(50) NOT NULL,
    uuid CHAR(36) NOT NULL,
    {entity}_id BIGINT NOT NULL,
    language VARCHAR(10) NOT NULL,
    {localized_field} VARCHAR(200),
    created_at DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP,
    created_by BIGINT,
    updated_at DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
    updated_by BIGINT,
    UNIQUE KEY uk_{entity}_i18n_lang ({entity}_id, language),
    KEY idx_{entity}_i18n_{entity}_id ({entity}_id),
    KEY idx_{entity}_i18n_language (language),
    CONSTRAINT fk_{entity}_i18n FOREIGN KEY ({entity}_id) REFERENCES {entities}(id) ON DELETE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci;

-- Migrate existing data
INSERT INTO {entity}_i18n (uid, uuid, {entity}_id, language, {field}, created_at, updated_at, created_by, updated_by)
SELECT CONCAT('{entity}i18n_', LPAD(id, 8, '0')), UUID(), id, 'TR', {field},
       COALESCE(created_at, NOW()), COALESCE(updated_at, NOW()), created_by, updated_by
FROM {entities} WHERE {field} IS NOT NULL;

ALTER TABLE {entities} DROP COLUMN {field};
```

### 5. Service Oluştur

```java
@Service @RequiredArgsConstructor @Transactional
public class {Entity}I18nServiceImpl implements {Entity}I18nService {
    private static final Language DEFAULT = Language.TR;
    private final {Entity}I18nRepository repo;

    @Transactional(readOnly = true)
    public {Entity}I18nResponse get(Long id, Language lang) {
        return {Entity}I18nResponse.from(repo.findBy{Entity}IdAndLanguage(id, lang)
            .orElseGet(() -> getFallback(id)));
    }

    public {Entity}I18nResponse upsert(Long id, Language lang, {Entity}I18nRequest req) {
        var i18n = repo.findBy{Entity}IdAndLanguage(id, lang)
            .orElseGet(() -> { var n = new {Entity}I18n(); n.set{Entity}Id(id); n.setLanguage(lang); return n; });
        if (req.{field}() != null) i18n.set{Field}(req.{field}());
        return {Entity}I18nResponse.from(repo.save(i18n));
    }

    private {Entity}I18n getFallback(Long id) {
        return repo.findBy{Entity}IdAndLanguage(id, DEFAULT).orElseGet(() -> {
            var all = repo.findBy{Entity}Id(id);
            if (all.isEmpty()) throw new EntityNotFoundException("{Entity}I18n", id);
            return all.get(0);
        });
    }
}
```
