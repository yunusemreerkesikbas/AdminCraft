#### 1.1 TypeScript Implementation (page-edit-dialog.component.ts)

```typescript
// Imports
import { PageTemplateService } from '../../templates/page-template.service';
import { PageTemplateDto } from '../../templates/page-template.types';

// Class properties (following ComponentEditDialog pattern)
readonly #templateService = inject(PageTemplateService);

// Signals for template data
protected pageTemplates = signal<PageTemplateDto[]>([]);
protected templatesLoading = signal<boolean>(false);

// Options for spa-select
templateOptions: { value: number | null; label: string }[] = [];

// In ngOnInit() - Load templates
override ngOnInit(): void {
    super.ngOnInit();
    this.#loadPageTemplates();
}

// Load templates method (following ComponentListComponent pattern)
#loadPageTemplates(): void {
    this.templatesLoading.set(true);
    this.#templateService.getActiveTemplates()
        .pipe(take(1))
        .subscribe({
            next: (templates) => {
                this.pageTemplates.set(templates);
                this.templateOptions = [
                    { value: null, label: this.#translocoService.translate('admin.pages.fields.noTemplate') },
                    ...templates.map(t => ({
                        value: t.id,
                        label: t.name
                    }))
                ];
                this.templatesLoading.set(false);
            },
            error: () => {
                this.#notify.alert('admin.pages.errors.loadTemplatesFailed');
                this.templatesLoading.set(false);
            }
        });
}

// In buildGeneralForm() - Add templateId
protected buildGeneralForm(): FormGroup {
    const page = this.data?.page;
    return this.fb.group({
        templateId: [page?.templateId || null],  // Optional, no validator
        status: [page?.status || 'DRAFT', Validators.required],
        styleClasses: [page?.styleClasses || ''],
        robotTag: [page?.robotTag || 'INDEX_FOLLOW', Validators.required],
        uid: [page?.uid || '']
    });
}
```

#### 2.2 Create Composite with Template Assignment

```java
@Override
@Transactional
public PageDetailResponse createComposite(CreatePageCompositeRequest request, Long userId) {
    // 1. Validate template exists (if provided)
    if (request.templateId() != null) {
        pageTemplateService.getById(request.templateId()); // Throws if not found
    }

    // 2. Create page
    Page page = Page.builder()
        .uid(generateUidIfEmpty(request.uid()))
        .templateId(request.templateId())
        .status(request.status() != null ? request.status() : PageStatus.DRAFT)
        .styleClasses(request.styleClasses())
        .robotTag(request.robotTag())
        .build();

    Page savedPage = pageRepository.save(page);

    // 3. Create translations
    createTranslations(savedPage, request.translations());

    // 4. Auto-create slots from template (NEW!)
    if (request.templateId() != null) {
        pageTemplateService.assignTemplateToPage(savedPage.getId(), request.templateId());
    }

    return toDetailResponse(savedPage);
}
```

#### 2.3 Update Composite with Template Change Handling

```java
@Override
@Transactional
public PageDetailResponse updateComposite(Long id, UpdatePageCompositeRequest request, Long userId) {
    Page page = pageRepository.findById(id)
        .orElseThrow(() -> new ResourceNotFoundException("Page not found: " + id));

    Long oldTemplateId = page.getTemplateId();
    Long newTemplateId = request.templateId();

    // 1. Validate new template (if provided)
    if (newTemplateId != null) {
        pageTemplateService.getById(newTemplateId); // Throws if not found
    }

    // 2. Update page fields
    if (newTemplateId != null) {
        page.setTemplateId(newTemplateId);
    }
    if (request.status() != null) {
        page.setStatus(request.status());
    }
    // ... other fields

    Page savedPage = pageRepository.save(page);

    // 3. Update translations
    if (request.translations() != null) {
        updateTranslations(savedPage, request.translations());
    }

    // 4. Handle template change - create missing slots (NEW!)
    if (newTemplateId != null && !newTemplateId.equals(oldTemplateId)) {
        pageTemplateService.assignTemplateToPage(savedPage.getId(), newTemplateId);
    }

    return toDetailResponse(savedPage);
}
```

#### 2.4 Existing assignTemplateToPage Logic (Reference)

```java
// Already exists in PageTemplateService - creates PageSlots from TemplateSlots
@Transactional
public void assignTemplateToPage(Long pageId, Long templateId) {
    PageTemplate template = getById(templateId);
    Page page = pageRepository.findById(pageId)
        .orElseThrow(() -> new ResourceNotFoundException("Page not found"));

    // Get existing page slot names
    Set<String> existingSlotNames = pageSlotRepository.findByPageId(pageId)
        .stream()
        .map(PageSlot::getSlotName)
        .collect(Collectors.toSet());

    // Create missing slots from template
    for (TemplateSlot ts : template.getSlots()) {
        if (!existingSlotNames.contains(ts.getSlotName())) {
            PageSlot pageSlot = PageSlot.builder()
                .page(page)
                .slotName(ts.getSlotName())
                .position(ts.getPosition())
                .sortOrder(ts.getSortOrder())
                .isActive(true)
                .isShared(false)
                .build();
            pageSlotRepository.save(pageSlot);
        }
    }
}
```

## ⚙️ Technical Details

### Template Service Integration

**Existing Endpoint:**

```
POST /api/page-templates/{templateId}/assign/{pageId}
```

**Approach A (Recommended):** Call from PageServiceImpl

- Single transaction
- Consistent behavior
- No extra HTTP call

**Approach B:** Call from frontend after page create

- Two HTTP calls
- Possible race condition
- More complex error handling

### Slot Inheritance Strategy

**Decision:** Runtime merge (current approach)

- Template slots loaded as shared
- Page slots override by name
- No allowOverwrite flag needed (simpler)

### CMS Response Strategy

**Decision:** Keep current flat format, add metadata

- Don't break existing consumers
- Add optional `template`, `slotShared` fields
- Backwards compatible
